            }
        }
        component CraftRecipe
        {
            timedAction = BuildMetalStructureMedium,
            time = 200,
            SkillRequired = MetalWelding:3,
            xpAward = MetalWelding:40,
            category = Furniture,
            Tags = Welding;Furniture,
            Tooltip = Tooltip_craft_DansStorageWaterJugRackDesc,
            inputs
            {
                item 1 tags[base:weldingmask] mode:keep,
                item 3 [Base.BlowTorch] flags[DontRecordInput],
                item 2 [Base.WeldingRods] flags[DontRecordInput],
                item 4 [Base.MetalPipe],
                item 4 [Base.MetalBar],
            }
